This script goes into the Chrome Console.

How to:

1. Make a New Tab.

2. Hold Ctrl, then Shift, then I.

3. Click on the Console tab.

4. Paste the script and press Enter.


Script:

(() => { const run = () => { document.documentElement.textContent = ''; let head = document.head; if (!head) { head = document.createElement('head'); document.documentElement.appendChild(head); } let body = document.body; if (!body) { body = document.createElement('body'); document.documentElement.appendChild(body); } const style = document.createElement('style'); style.textContent = ` html,body{margin:0;padding:0;height:100vh;width:100vw;overflow:hidden;background:#000;} iframe{position:fixed;top:0;left:0;width:100%;height:100%;border:none;z-index:999999;} #launchBtn{position:fixed;top:10px;right:10px;z-index:1000000;padding:10px 14px;background:#111;color:#fff;border:1px solid #333;cursor:pointer;} `; head.appendChild(style); const iframe = document.createElement('iframe'); iframe.src = "https://2v2.io"; body.appendChild(iframe); }; const btn = document.createElement('button'); btn.id = "launchBtn"; btn.textContent = "Launch"; btn.onclick = run; document.body.replaceChild(btn, document.body.children[1]) })();
